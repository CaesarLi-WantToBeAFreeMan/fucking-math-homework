## 一

1. $f(x - 3) = 2{x}^{2} - x + 1$

$$
    \begin{align*}
        &\text {set } t = x - 3\\
        &\therefore x = t + 3\\
        f(t) &= 2({t + 3})^{2} - (t + 3) + 1\\
        &= 2({t}^{2} + 6t + 9) - (t + 3) + 1\\
        &= 2{t}^{2} + 11t + 16\\
        &\therefore f(x) = 2{x}^{2} + 11x + 16
    \end{align*}
$$

2. $f(\frac {1} {x}) = \frac {x} {x + 1}$

$$
    \begin{align*}
        &\text {set } t = \frac {1} {x}\\
        &\therefore x = \frac {1} {t}\\
        f(t) &= \frac {\frac {1} {t}} {\frac {1} {t} + 1}\\
        &= \frac {1} {1 + t}\\
        &\therefore f(x) = \frac {1} {1 + x}
    \end{align*}
$$

3. $f(2x + 1) = 4{x}^{2} + 6x$

$$
    \begin{align*}
        &\text {set } t = 2x + 1\\
        &\therefore x = \frac {t - 1} {2}\\
        f(t) &= 4({\frac {t - 1} {2}})^{2} + 6(\frac {t - 1} {2})\\
        &= 4(\frac {{t}^{2} - 2t + 1} {4}) + 3(t - 1)\\
        &= {t}^{2} - 2t + 1 + 3t - 3\\
        &= {t}^{2} + t - 2\\
        &\therefore f(x) = {x}^{2} + x - 2
    \end{align*}
$$

4. $f(x + \frac {1} {x}) = {x}^{2} + \frac {1} {{x}^{2}} + 3$

$$
    \begin{align*}
        &\because f(x + \frac {1} {x})^{2} = {x}^{2} + 2 + \frac {1} {{x}^{2}}\\
        &\therefore f(x) = {x}^{2} + 1
    \end{align*}
$$

5. $f(\sqrt {x} - 1) = x - 2 \sqrt {x}$

$$
    \begin{align*}
        &\because f(\sqrt {x} - 1)^{2} = x - 2 \sqrt {x} + 1\\
        &\therefore f(x) = x - 1
    \end{align*}
$$

6. $f({e}^{x}) = x + 2$

$$
    \begin{align*}
        &\text {set } t = {e}^{x}\\
        &\therefore x = \ln {t}\\
        f(t) &= \ln {t} + 2\\
        &\therefore f(x) = \ln {x} + 2
    \end{align*}
$$

## 二

1. $y = 3x - 5$

$$
    \begin{align*}
        &\because x = \frac {y + 5} {3}\\
        &\therefore {y}^{-1} = \frac {x + 5} {3}
    \end{align*}
$$

2. $y = \frac {2x + 1} {x - 1} (x \neq 1)$

$$
    \begin{align*}
        &\because x = \frac {y + 1} {y - 2}\\
        &\therefore y = \frac {x + 1} {x - 2}
    \end{align*}
$$

3. $y = {x}^{3} + 2$

$$
    \begin{align*}
        &\because x = \sqrt [3] {y - 2}\\
        &\therefore y = \sqrt [3] {x - 2}
    \end{align*}
$$

4. $y = \sqrt {x + 1} (x \geq -1)$

$$
    \begin{align*}
        &\because x = {y}^{2} - 1\\
        &\therefore y = {x}^{2} - 1
    \end{align*}
$$

5. $y = \frac {1} {{2}^{x} + 1}$

$$
    \begin{align*}
        &\because x = \log_2 (\frac {1} {y} - 1)\\
        &\therefore y = \log_2 (\frac {1} {y} - 1)
    \end{align*}
$$
